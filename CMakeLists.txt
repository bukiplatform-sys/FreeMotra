cmake_minimum_required(VERSION 3.24)
project(FreeMotra VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "FreeMotra requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Gui Qml Quick QuickControls2 Test)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_library(freemotra_core STATIC src/Identity.cpp src/Identity.h src/SessionGate.cpp src/SessionGate.h)
target_link_libraries(freemotra_core PUBLIC Qt6::Core PRIVATE Qt6::Network bcrypt)
target_include_directories(freemotra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(freemotra_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
qt_add_executable(FreeMotra WIN32 src/main.cpp src/AppController.cpp src/AppController.h src/windows/DesktopCapture.cpp src/windows/DesktopCapture.h src/windows/InputController.cpp src/windows/InputController.h)
target_link_libraries(FreeMotra PRIVATE freemotra_core Qt6::Gui Qt6::Quick Qt6::QuickControls2 d3d11 dxgi user32)
target_compile_definitions(FreeMotra PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(freemotra_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(FreeMotra PRIVATE /W4 /utf-8 /permissive-)
endif()
qt_add_qml_module(FreeMotra URI FreeMotra VERSION 1.0 QML_FILES
  qml/Main.qml qml/ActionButton.qml qml/Panel.qml qml/Glyph.qml qml/Field.qml)
set(CMAKE_INSTALL_BINDIR ".")
include(GNUInstallDirs)
install(TARGETS FreeMotra RUNTIME DESTINATION .)
qt_generate_deploy_qml_app_script(TARGET FreeMotra OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
install(DIRECTORY docs/ DESTINATION docs)
install(FILES LICENSE NOTICE.md DESTINATION .)
enable_testing()
qt_add_executable(core_tests tests/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE freemotra_core Qt6::Test)
add_test(NAME core_security COMMAND core_tests)
add_test(NAME qml_startup COMMAND FreeMotra --smoke-test)
set_tests_properties(qml_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 25)
set(CPACK_PACKAGE_NAME FreeMotra)
set(CPACK_PACKAGE_VENDOR FreeMotra)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY FreeMotra)
set(CPACK_PACKAGE_FILE_NAME "FreeMotra-${PROJECT_VERSION}-windows-x64")
set(CPACK_GENERATOR "ZIP;NSIS")
set(CPACK_NSIS_DISPLAY_NAME "FreeMotra 0.1")
set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
set(CPACK_PACKAGE_EXECUTABLES "FreeMotra;FreeMotra")
set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
include(CPack)
