cmake_minimum_required(VERSION 3.24)
project(FreeMotra VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "FreeMotra requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Gui Qml Quick QuickControls2 Multimedia Concurrent Test)
find_package(LibDataChannel CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_library(freemotra_core STATIC src/Identity.cpp src/Identity.h src/SessionGate.cpp src/SessionGate.h)
target_link_libraries(freemotra_core PUBLIC Qt6::Core PRIVATE Qt6::Network bcrypt)
target_include_directories(freemotra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(freemotra_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
qt_add_library(freemotra_transport STATIC src/LanLink.cpp src/LanLink.h)
target_link_libraries(freemotra_transport PUBLIC Qt6::Network LibDataChannel::LibDataChannel PRIVATE OpenSSL::Crypto crypt32)
target_compile_definitions(freemotra_transport PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(freemotra_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
qt_add_library(freemotra_session STATIC src/SessionTools.cpp src/SessionTools.h)
target_link_libraries(freemotra_session PUBLIC freemotra_transport Qt6::Multimedia Qt6::Concurrent)
qt_add_executable(FreeMotra WIN32 src/main.cpp src/SponsorFeed.cpp src/SponsorFeed.h src/AppController.cpp src/AppController.h src/windows/DesktopCapture.cpp src/windows/DesktopCapture.h src/windows/InputController.cpp src/windows/InputController.h)
target_link_libraries(FreeMotra PRIVATE freemotra_core freemotra_session freemotra_transport Qt6::Network LibDataChannel::LibDataChannel OpenSSL::Crypto Qt6::Gui Qt6::Quick Qt6::QuickControls2 d3d11 dxgi user32)
target_compile_definitions(FreeMotra PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(freemotra_session PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(freemotra_transport PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(freemotra_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(FreeMotra PRIVATE /W4 /utf-8 /permissive-)
endif()
qt_add_qml_module(FreeMotra URI FreeMotra VERSION 1.0 QML_FILES
  qml/Main.qml qml/SessionPanel.qml qml/RemoteWindow.qml qml/ActionButton.qml qml/Panel.qml qml/Glyph.qml qml/Field.qml)
include(GNUInstallDirs)
install(TARGETS FreeMotra RUNTIME DESTINATION bin)
qt_generate_deploy_qml_app_script(TARGET FreeMotra OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
install(FILES "${Qt6_DIR}/../../../plugins/platforms/qoffscreen.dll" DESTINATION plugins/platforms)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES LICENSE NOTICE.md DESTINATION .)
if(DEFINED VCPKG_INSTALLED_DIR)
  install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/" DESTINATION third-party FILES_MATCHING PATTERN "copyright")
endif()
enable_testing()
qt_add_executable(lan_tests tests/LanTests.cpp)
target_link_libraries(lan_tests PRIVATE freemotra_transport Qt6::Test)
add_test(NAME lan_transport COMMAND lan_tests)
set_tests_properties(lan_transport PROPERTIES TIMEOUT 100)
qt_add_executable(quality_tests tests/QualityTests.cpp)
target_include_directories(quality_tests PRIVATE src)
target_link_libraries(quality_tests PRIVATE Qt6::Test)
add_test(NAME adaptive_quality COMMAND quality_tests)
qt_add_executable(session_tests tests/SessionTests.cpp)
target_link_libraries(session_tests PRIVATE freemotra_session Qt6::Test)
add_test(NAME session_features COMMAND session_tests)
qt_add_executable(protocol_tests tests/SessionTests.cpp tests/InMemoryLink.cpp)
target_link_libraries(protocol_tests PRIVATE freemotra_session Qt6::Test)
add_test(NAME session_protocol_in_memory COMMAND protocol_tests)
set_tests_properties(session_features PROPERTIES TIMEOUT 180)
qt_add_executable(core_tests tests/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE freemotra_core Qt6::Test)
add_test(NAME core_security COMMAND core_tests)
add_test(NAME qml_startup COMMAND FreeMotra --smoke-test)
set_tests_properties(qml_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 25)
set(CPACK_PACKAGE_NAME FreeMotra)
set(CPACK_PACKAGE_VENDOR FreeMotra)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY FreeMotra)
set(CPACK_PACKAGE_FILE_NAME "FreeMotra-${PROJECT_VERSION}-windows-x64")
set(CPACK_GENERATOR "ZIP;NSIS")
set(CPACK_NSIS_DISPLAY_NAME "FreeMotra 0.3")
set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
set(CPACK_PACKAGE_EXECUTABLES "FreeMotra;FreeMotra")
set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION "bin")
include(InstallRequiredSystemLibraries)
include(CPack)
